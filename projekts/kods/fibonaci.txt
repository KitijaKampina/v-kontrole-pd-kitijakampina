#include <iostream>
#include <bits/stdc++.h>
using namespace std;
int main()
{
    vector<unsigned long long>fib(100);
    fib[0]=0;
    fib[1]=1;
    for(int i=2; i<100; i++)
        fib[i]=fib[i-1]+fib[i-2];
    reverse(fib.begin(), fib.end());
    for(auto x : fib)
        cout << x << ' ';
    cout << endl;
    return 0;
}
